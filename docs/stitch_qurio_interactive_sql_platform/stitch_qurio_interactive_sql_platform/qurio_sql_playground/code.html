-- Análisis de retención mensual y volumen de compras WITH resumen_mensual AS ( SELECT DATE_TRUNC('month', o.creado_en) AS mes_compra, COUNT(DISTINCT o.cliente_id) AS clientes_activos, SUM(o.monto_total) AS ingresos_totales, ROUND(AVG(o.monto_total), 2) AS ticket_promedio FROM ordenes o WHERE o.estado = 'completada' GROUP BY DATE_TRUNC('month', o.creado_en) ) SELECT TO_CHAR(mes_compra, 'YYYY-MM') AS periodo, clientes_activos, CONCAT('$', TO_CHAR(ingresos_totales, 'FM999,999,990')) AS ingresos, ticket_promedio FROM resumen_mensual ORDER BY mes_compra DESC LIMIT 6;
